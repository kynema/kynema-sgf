43 static void constant_chord(ComponentView& data, FLLCData& fllc)
44 {
45 const int npts = static_cast<int>(fllc.correction_velocity.size());
46 auto& du = fllc.correction_velocity;
47
48 auto& u_les = fllc.les_velocity;
49 auto& u_opt = fllc.optimal_velocity;
50 auto& G = fllc.lift;
51 auto& dG = fllc.grad_lift;
52
58 for (int ip = 0; ip < npts; ++ip) {
59 const auto force = data.force[ip];
60 const auto vel = data.vel_rel[ip];
61 const auto dr = fllc.dr[ip];
62 const auto vmag = amrex::max<amrex::Real>(
63 vs::mag(vel), vs::DTraits<amrex::Real>::eps());
64 const auto vmag2 = vmag * vmag;
65
66 const auto fv = force & vel;
67
68 G[ip] = (force - vel * fv / vmag2) / dr;
69 }
70
77 dG[0] = G[0];
78 dG[npts - 1] = -1 * G[npts - 1];
79 for (int ip = 1; ip < npts - 1; ++ip) {
80 dG[ip] = 0.5_rt * (G[ip + 1] - G[ip - 1]);
81 }
82
89 for (int ip = 0; ip < npts; ++ip) {
90
91 // The force is spread with the largest of epsilon, epsilon_min
92 // (turbines; fllc.epsilon is the larger of the two) and the
93 // optimal epsilon, so that is the width the simulation resolves
94 const auto eps_opt = fllc.optimal_epsilon[ip];
95 const auto eps_les = amrex::max<amrex::Real>(fllc.epsilon, eps_opt);
96
97 for (int jp = 0; jp < npts; ++jp) {
98
99 if (ip == jp) {
100 continue;
101 }
102
103 const auto r_dis = vs::mag(data.pos[ip] - data.pos[jp]);
104 const auto& vel = data.vel_rel[jp];
105 const auto vmag = amrex::max<amrex::Real>(
106 vs::mag(vel), vs::DTraits<amrex::Real>::eps());
107
108 auto coefficient =
109 1.0_rt /
110 (-4.0_rt * std::numbers::pi_v<amrex::Real> * r_dis * vmag);
111 const auto cLes =
112 1.0_rt - std::exp(-r_dis * r_dis / (eps_les * eps_les));
113 const auto cOpt =
114 1.0_rt - std::exp(-r_dis * r_dis / (eps_opt * eps_opt));
115
116 // The sign of the induced velocity depends on which side of the
117 // blade we are on
118 if (ip < jp) {
119 coefficient *= -1.0_rt;
120 }
121
122 u_les[ip] = u_les[ip] - dG[jp] * coefficient * cLes;
123 u_opt[ip] = u_opt[ip] - dG[jp] * coefficient * cOpt;
124 }
125
126 // Relaxation to compute the induced velocity
127 const auto f = fllc.relaxation_factor;
128 du[ip] = (1.0_rt - f) * du[ip] + f * (u_opt[ip] - u_les[ip]);
129 }
130
136 add_correction(data, fllc);
137 for (int ip = 0; ip < npts; ++ip) {
138 u_les[ip] *= 0.0_rt;
139 u_opt[ip] *= 0.0_rt;
140 }
141 }

151 {

153

177 static amrex::Real kernel(const amrex::Real r_dis2, const amrex::Real eps2)
178 {
179 const auto x = -r_dis2 / eps2;
180 if (x == 0.0_rt) {
181 return 0.5_rt / eps2;
182 }
183 return (std::exp(x) / eps2) + (std::expm1(x) / (2.0_rt * r_dis2));
184 }

219 interp::linear(

221 fllc.nonuniform_optimal_epsilon);
222 interp::linear(

224 }
225
226 const auto& optimal_epsilon = (fllc.nonuniform)
227 ? fllc.nonuniform_optimal_epsilon
228 : fllc.optimal_epsilon;
229 const auto& dr = (fllc.nonuniform) ? fllc.nonuniform_dr : fllc.dr;
230 const auto& vel_rel =
231 (fllc.nonuniform) ? fllc.nonuniform_vel_rel : fllc.vel_rel;
232 const auto& G_new = (fllc.nonuniform) ? nonuniform_G : G;
233
238 for (int ip = 0; ip < npts; ++ip) {
239
240 const auto coefficient =
241 1.0_rt / (2.0_rt * std::numbers::pi_v<amrex::Real>);
242
243 for (int jp = 0; jp < dr.size(); ++jp) {
244
245 // The force is spread with the largest of epsilon,
246 // epsilon_min (turbines; fllc.epsilon is the larger of the
247 // two) and the optimal epsilon, so that is the width the
248 // simulation resolves

250 const auto eps_les =

252 const auto eps_les2 = eps_les * eps_les;
253 const auto eps_opt2 = eps_opt * eps_opt;
254 const auto& vel = vel_rel[jp];
255 const auto vmag = amrex::max<amrex::Real>(
256 vs::mag(vel), vs::DTraits<amrex::Real>::eps());
257
258 const auto r_dis =

261 : vs::mag(data.pos[ip] - data.pos[jp]);
262 const auto r_dis2 = r_dis * r_dis;
263 const auto k_les = kernel(r_dis2, eps_les2);
264 const auto k_opt = kernel(r_dis2, eps_opt2);
265
266 u_les[ip] =
267 u_les[ip] + coefficient * G_new[jp] / vmag * k_les * dr[jp];
268 u_opt[ip] =
269 u_opt[ip] + coefficient * G_new[jp] / vmag * k_opt * dr[jp];
270 }
271

273 const auto f = fllc.relaxation_factor;

275 }
