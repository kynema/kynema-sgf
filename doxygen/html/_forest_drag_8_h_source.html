65 // Compute the type-2 peak LAD scaling factor from integrated LAI.
66 AMREX_GPU_DEVICE AMREX_FORCE_INLINE amrex::Real lm() const
67 {
68 amrex::Real treelaimax = 0.0_rt;
69 if (m_type_forest == 2) {
70 const amrex::Real captured_height_forest = m_height_forest;
71 const amrex::Real treeZm = m_laimax_forest * captured_height_forest;
72 const int n = 100;
73 const amrex::Real int_0_fh = kynema_sgf::utils::trapz(
74 0, captured_height_forest - constants::TIGHT_TOL, n,
75 [=](const amrex::Real x) {
76 const amrex::Real ratio =
77 (captured_height_forest - treeZm) /
78 (captured_height_forest - x);
79 const auto exponent = (x < treeZm) ? 6.0_rt : 0.5_rt;
80 return std::pow(ratio, exponent) *
81 std::exp(exponent * (1 - ratio));
82 });
83 treelaimax = m_lai_forest / int_0_fh;
84 }
85 return treelaimax;
86 }

88 // Vertical LAD profile used by legacy cylindrical forests.
89 AMREX_GPU_DEVICE AMREX_FORCE_INLINE amrex::Real
90 area_fraction(const amrex::Real z, const amrex::Real treelaimax) const
91 {
92 amrex::Real af = 0.0_rt;
93 if (m_type_forest == 1) {
94 af = m_lai_forest / m_height_forest;
95 } else if (m_type_forest == 2 && z < m_height_forest) {
96 // The profile tends to zero at the tree top, where the ratio
97 // below divides by zero, so the top and above are left at zero.
98 const auto treeZm = m_laimax_forest * m_height_forest;
99 const auto ratio =
100 (m_height_forest - treeZm) / (m_height_forest - z);
101 const auto exponent = (z < treeZm) ? 6.0_rt : 0.5_rt;
102 af = treelaimax * std::pow(ratio, exponent) *
103 std::exp(exponent * (1 - ratio));
104 }
105 return af;
106 }

108 // Convex-hull half-space test (boundary-inclusive) using precomputed
109 // inward edge normals.
110 AMREX_GPU_DEVICE AMREX_FORCE_INLINE bool point_in_on_hull(
111 const amrex::Real px,
112 const amrex::Real py,
113 const ForestHullEdge* hull_edges) const
114 {
115 if (m_hull_edge_count <= 0) {
116 return true;
117 }
118 for (int i = 0; i < m_hull_edge_count; ++i) {
119 const auto& e = hull_edges[m_hull_edge_offset + i];
120 const amrex::Real signed_dist =
121 (e.m_nx * px) + (e.m_ny * py) - e.m_d;
122 if (signed_dist < -constants::TIGHT_TOL * e.m_tol_scale) {
123 return false;
124 }
125 }
126 return true;
127 }

129 // Return real-space bounds for either point-cloud or cylindrical mode.
130 AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE amrex::RealBox real_bounding_box(
131 const amrex::GpuArray<amrex::Real, AMREX_SPACEDIM>& prob_lo) const
132 {
133 if (m_drag_mode == 1) {
134 return {m_bbox_xlo, m_bbox_ylo, m_bbox_zlo + m_terrain_zmin,
135 m_bbox_xhi, m_bbox_yhi, m_bbox_zhi + m_terrain_zmax};
136 }
137
138 const amrex::Real search_tol = 1.1_rt;
139 const amrex::Real search_radius =
140 0.5_rt * search_tol * m_diameter_forest;
141 const auto x0 = m_x_forest - search_radius;
142 const auto y0 = m_y_forest - search_radius;
143 const auto z0 = prob_lo[2];
144 const auto x1 = m_x_forest + search_radius;
145 const auto y1 = m_y_forest + search_radius;
146 const auto z1 = m_height_forest + m_terrain_zmax;
147 return {x0, y0, z0, x1, y1, z1};
148 }
149
150 // Integer box used for tile intersection tests before launching kernels.
151 [[nodiscard]] amrex::Box bounding_box(const amrex::Geometry& geom) const
152 {
153 return utils::realbox_to_box(
154 real_bounding_box(geom.ProbLoArray()), geom);
155 }

176AMREX_GPU_DEVICE AMREX_FORCE_INLINE bool cylinder_lad(
177 const Forest& fst,
178 const amrex::Real x,
179 const amrex::Real y,
180 const amrex::Real zagl,
181 amrex::Real& lad)
182{
183 lad = 0.0_rt;
184 const auto radius = std::sqrt(
185 ((x - fst.m_x_forest) * (x - fst.m_x_forest)) +
186 ((y - fst.m_y_forest) * (y - fst.m_y_forest)));
187 if (zagl <= fst.m_height_forest &&
188 radius <= (0.5_rt * fst.m_diameter_forest)) {
189 lad = fst.area_fraction(zagl, fst.lm());
190 return true;
191 }
192 return false;
193}

208AMREX_GPU_DEVICE AMREX_FORCE_INLINE amrex::Real point_cloud_lad(
209 const Forest& fst,
210 const ForestPoint* points,
211 const ForestHullEdge* hull_edges,
212 const amrex::Real x,
213 const amrex::Real y,
214 const amrex::Real z,
215 const amrex::Real half_dz,
216 const int num_neighbors,
217 const amrex::Real interp_eps)
218{
219 if (fst.m_cloud_point_count <= 0 ||
220 !fst.point_in_on_hull(x, y, hull_edges)) {
221 return 0.0_rt;
222 }
223
224 constexpr int max_neighbors = 8;
225 constexpr amrex::Real huge = constants::LARGE_NUM;
226 amrex::GpuArray<amrex::Real, max_neighbors> nearest_d2;
227 amrex::GpuArray<amrex::Real, max_neighbors> nearest_lad;
228 amrex::GpuArray<amrex::Real, max_neighbors> nearest_z;
229 for (int n = 0; n < max_neighbors; ++n) {
230 nearest_d2[n] = huge;
231 nearest_lad[n] = 0.0_rt;
232 nearest_z[n] = 0.0_rt;
233 }
234
235 // Sorted nearest-neighbor list, updated in place.
236 const int pstart = fst.m_cloud_point_offset;
237 const int pend = pstart + fst.m_cloud_point_count;
238 for (int p = pstart; p < pend; ++p) {
239 const auto& pt = points[p];
240 const auto dxp = x - pt.m_x;
241 const auto dyp = y - pt.m_y;
242 const auto dzp = z - pt.m_z;
243 const auto d2 = (dxp * dxp) + (dyp * dyp) + (dzp * dzp);
244 if (d2 < nearest_d2[num_neighbors - 1]) {
245 int insert = num_neighbors - 1;
246 while (insert > 0 && d2 < nearest_d2[insert - 1]) {
247 nearest_d2[insert] = nearest_d2[insert - 1];
248 nearest_lad[insert] = nearest_lad[insert - 1];
249 nearest_z[insert] = nearest_z[insert - 1];
250 --insert;
251 }
252 nearest_d2[insert] = d2;
253 nearest_lad[insert] = pt.m_lad;
254 nearest_z[insert] = pt.m_z;
255 }
256 }
257
258 // No LAD above the highest of the nearest samples.
259 amrex::Real max_z_neighbors = 0.0_rt;
260 for (int n = 0; n < num_neighbors; ++n) {
261 max_z_neighbors =
262 amrex::max<amrex::Real>(max_z_neighbors, nearest_z[n]);
263 }
264
265 const auto eps2 = interp_eps * interp_eps;
266 amrex::Real lad_interp = 0.0_rt;
267 if (nearest_d2[0] < eps2) {
268 lad_interp = nearest_lad[0];
269 } else if (z - half_dz <= max_z_neighbors) {
270 amrex::Real sum_w = 0.0_rt;
271 amrex::Real sum_lad = 0.0_rt;
272 for (int n = 0; n < num_neighbors; ++n) {
273 if (nearest_d2[n] >= huge) {
274 continue;
275 }
276 const auto w = 1.0_rt / std::sqrt(nearest_d2[n] + eps2);
277 sum_w += w;
278 sum_lad += w * nearest_lad[n];
279 }
280 if (sum_w > 0.0_rt) {
281 lad_interp = sum_lad / sum_w;
282 }
283 }
284 return lad_interp;
285}
