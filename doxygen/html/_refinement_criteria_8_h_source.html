53 const bool previous_val, const bool current_val, const TaggingOperator& op)
54{
55 switch (op) {
56 case TaggingOperator::AND:
57 return (previous_val && current_val) ? amrex::TagBox::SET
58 : amrex::TagBox::CLEAR;
59 case TaggingOperator::OR:
60 return (previous_val || current_val) ? amrex::TagBox::SET
61 : amrex::TagBox::CLEAR;
62 case TaggingOperator::AND_NOT:

64 : amrex::TagBox::CLEAR;
