22AMREX_FORCE_INLINE int closest_index(
23 const amrex::Vector<amrex::Real>& vec,
24 const amrex::Real value,
25 const amrex::Real tol = 0.0_rt)
26{
27 auto it = std::ranges::upper_bound(vec, value);
28 if (it == vec.end()) {
29 // Try again with tolerance added
30 it = std::ranges::upper_bound(vec, value - tol);
31 }
32 AMREX_ALWAYS_ASSERT(it != vec.end());
33
34 const int idx = static_cast<int>(std::distance(vec.begin(), it));
35 return amrex::max(idx - 1, 0);
36}

40AMREX_FORCE_INLINE T perpendicular_idx(const int normal)
41{
42 switch (normal) {
43 case 0:
44 return T{1, 2};
45 case 1:

47 case 2:

68AMREX_INLINE amrex::Box face_aware_boundary_box_intersection(
69 amrex::IntVect& shift_to_cc,
70 amrex::Box grown_interior_box,
71 const amrex::Box& domain_boundary_box,
72 const amrex::Orientation& ori)
73{
74 shift_to_cc = {0, 0, 0};
75 const auto& field_location_vector = grown_interior_box.type();
76 if (!grown_interior_box.cellCentered()) {
77 grown_interior_box.enclosedCells();
78 }
79 auto bx = grown_interior_box & domain_boundary_box;
80 if (bx.isEmpty()) {
81 return bx;
82 }
83
84 if (ori.isHigh() && field_location_vector[ori.coordDir()] == 1) {
85 bx.shift(field_location_vector);
86 shift_to_cc = -field_location_vector;
87 }
88 return bx;
89}

116AMREX_INLINE void extend_boundary_ghost_layers(
117 const amrex::Box& bx,
118 amrex::Array4<amrex::Real> const& arr,
119 const amrex::Orientation& ori,
120 const int scomp,
121 const int ncomp)
122{
123 if (!bx.ok()) {
124 return;
125 }
126 const int idir = ori.coordDir();
127 AMREX_ASSERT(bx.length(idir) == 1);
128 const amrex::IntVect arr_lo(amrex::lbound(arr));
129 const amrex::IntVect arr_hi(amrex::ubound(arr));
130 const int nlayers = ori.isLow() ? (bx.smallEnd(idir) - arr_lo[idir])
131 : (arr_hi[idir] - bx.bigEnd(idir));
132 if (nlayers <= 0) {
133 return;
134 }
135 const amrex::IntVect outward =
136 amrex::IntVect::TheDimensionVector(idir) * (ori.isLow() ? -1 : 1);
137 amrex::ParallelFor(
138 bx, ncomp, [=] AMREX_GPU_DEVICE(int i, int j, int k, int n) {
139 const amrex::IntVect iv{i, j, k};
140 for (int g = 1; g <= nlayers; ++g) {
141 arr(iv + (g * outward), scomp + n) = arr(iv, scomp + n);
142 }
143 });
144}

175AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE bool contains(
176 const amrex::Box& box,
177 const amrex::RealVect& pos,
178 const amrex::GpuArray<amrex::Real, AMREX_SPACEDIM>& prob_lo,
179 const amrex::GpuArray<amrex::Real, AMREX_SPACEDIM>& dxinv)
180{
181 const amrex::IntVect iv(AMREX_D_DECL(

186 return box.contains(iv);
