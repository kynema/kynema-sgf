15template <typename datatype>
16void read_ops(datatype& data, const utils::ActParser& pp)
17{
18 // Tell reading utility the name of the solver
19 data.meta().solver_name = data.meta().ext_ptr->identifier();
20 // Data common to any turbine actuator simulation
21 utils::read_inputs(data.meta(), data.info(), pp);
22
23 auto& tdata = data.meta();
24
25 // Get density value for normalization
26 pp.get("density", tdata.density);
27 // The forces of the external solver are divided by it
28 if (!(tdata.density > 0.0_rt)) {
29 amrex::Abort(
30 "Actuator " + data.info().label +
31 ": density must be larger than 0. Value read: " +
32 std::to_string(tdata.density));
33 }
34
35 // Copy data to external data holder
36 const auto& tinfo = data.info();
37 auto& tf = data.meta().ext_data;
38 for (int i = 0; i < AMREX_SPACEDIM; ++i) {
39 tf.base_pos[i] = static_cast<float>(tinfo.base_pos[i]);
40 }
41
42 tf.tlabel = tinfo.label;
43 tf.tid_global = tinfo.id;
44 tf.num_blades = tdata.num_blades;
45 tf.num_pts_blade = tdata.num_pts_blade;
46 tf.num_pts_tower = tdata.num_pts_tower;
47 tf.dt_cfd = data.sim().time().delta_t();
48}

50template <typename datatype>
51void external_determine_influenced_procs(datatype& data)
52{
53 auto& info = data.info();
54 info.procs =
55 utils::determine_influenced_procs(data.sim().mesh(), info.bound_box);
56
57 AMREX_ALWAYS_ASSERT_WITH_MESSAGE(
58 info.root_proc > -1,
59 "info.root_proc = " + std::to_string(info.root_proc));
60 // During regrid, the influenced processes might have changed and might
61 // no longer include the root proc. We insert it back to ensure that it
62 // is always present on the list.
63 info.procs.insert(info.root_proc);
64
65 const int iproc = amrex::ParallelDescriptor::MyProc();
66 auto in_proc = info.procs.find(iproc);
67 info.actuator_in_proc = (in_proc != info.procs.end());
68 info.sample_vel_in_proc = info.is_root_proc;
69}

72void external_determine_root_proc(
73 datatype& data, amrex::Vector<int>& act_proc_count)
74{
75 namespace utils = ::kynema_sgf::actuator::utils;
76 auto& info = data.info();
77 info.procs =
78 utils::determine_influenced_procs(data.sim().mesh(), info.bound_box);
79
80 utils::determine_root_proc(info, act_proc_count);
81
82 // TODO: This function is doing a lot more than advertised by the name.
83 // Should figure out a better way to perform the extra work.
84
85 // For OpenFAST we only need velocities sampled in root process
86 info.sample_vel_in_proc = info.is_root_proc;
87
88 // Initialize the OpenFAST object and register this turbine in the root

91 auto& tdata = data.meta();

97 tdata.ext_ptr->register_turbine(tdata.ext_data);
98 }

101template <typename datatype>
102void init_data_op(datatype& data)
103{
104
105 const auto& info = data.info();
106 auto& tdata = data.meta();
107
108 // Initialize our communicator for broadcasting data
109 amrex::ParallelDescriptor::Comm_dup(
110 amrex::ParallelDescriptor::Communicator(), tdata.tcomm);
111
112 amrex::Array<int, 5> sz_info = {0, 0, 0, 0, 0};
113 if (info.is_root_proc) {
114 tdata.ext_ptr->init_turbine(tdata.ext_data.tid_local);
115
116 const auto& tf = tdata.ext_data;
117 sz_info[0] = tf.num_blades;
118 sz_info[1] = tf.length_force(0);
119 sz_info[2] = tf.length_fluid_velocity(0);
120 sz_info[3] = tf.num_pts_tower;
121 sz_info[4] = tf.num_vel_pts_blade();
122 }
123
124 // Broadcast data to everyone
125 amrex::ParallelDescriptor::Bcast(
126 sz_info.begin(), sz_info.size(), info.root_proc, tdata.tcomm);
127
128 {
129 tdata.num_blades = sz_info[0];
130 // FLLC data sized from the num_blades input: one per blade of the
131 // external solver
132 if (!tdata.fllc.empty()) {
133 tdata.fllc.resize(tdata.num_blades, tdata.fllc.front());
134 }
135 // back calculate what the value per blade for number of points in
136 // the openfast data structure
137 tdata.num_vel_pts_blade = sz_info[4];
138 data.grid().resize(sz_info[1], sz_info[2]);
139 tdata.chord.resize(sz_info[1]);
140 tdata.num_pts_tower = sz_info[3];
141 }
142
143 tdata.vel_rel.assign(sz_info[1], vs::Vector::zero());
144
145 if (info.is_root_proc) {
146 // copy chord data
147 int npts = sz_info[1];
148 for (int i = 0; i < npts; ++i) {
149 tdata.chord[i] =
150 static_cast<amrex::Real>(tdata.ext_data.chord_at_force()[i]);
151 }
152 }
153
154 amrex::ParallelDescriptor::Bcast(
155 tdata.chord.data(), tdata.chord.size(), info.root_proc, tdata.tcomm);
156
157 make_component_views<datatype>(data);
158 init_epsilon<datatype>(data);
159}

161template <typename datatype>
162void update_pos_op(datatype& data)
163{
164 // Return early if this is not the root process for this turbine
165 //
166 // This is handled in Actuator class, but we add a check here just as a
167 // safeguard
168 if (!data.info().is_root_proc) {
169 return;
170 }
171
172 const auto& tdata = data.meta();
173 const auto& bp = data.info().base_pos;
174 const auto& pxvel = tdata.ext_data.position_at_vel(0);
175 const auto& pyvel = tdata.ext_data.position_at_vel(1);
176 const auto& pzvel = tdata.ext_data.position_at_vel(2);
177 auto& vel_pos = data.grid().vel_pos;
178 for (int i = 0; i < vel_pos.size(); ++i) {
179 vel_pos[i].x() = static_cast<amrex::Real>(pxvel[i]) + bp.x();
180 vel_pos[i].y() = static_cast<amrex::Real>(pyvel[i]) + bp.y();
181 vel_pos[i].z() = static_cast<amrex::Real>(pzvel[i]) + bp.z();
182 }
183}

185template <typename datatype>
186void update_fields_op(datatype& data)
187{
188 // Return early if this is not the root process for this turbine
189 //
190 // This is handled in Actuator class, but we add a check here just as a
191 // safeguard
192 if (!data.info().is_root_proc) {
193 return;
194 }
195 auto& tdata = data.meta();
196
197 const auto& uvel = tdata.ext_data.fluid_velocity(0);
198 const auto& vvel = tdata.ext_data.fluid_velocity(1);
199 const auto& wvel = tdata.ext_data.fluid_velocity(2);
200 const auto& vel = data.grid().vel;
201
202 if (!tdata.fllc.empty()) {
203 namespace interp = ::kynema_sgf::interp;
204
205 // The relative velocity needed for the FLLC is the fluid velocity
206 // last sent to the external solver minus the velocity of the blade.
207 // The first is known at the velocity points and the second at the
208 // force points, each with its own numbering: the hub first, then
209 // the points of every blade.
210 const auto& xdot = tdata.ext_data.solid_velocity(0);
211 const auto& ydot = tdata.ext_data.solid_velocity(1);
212 const auto& zdot = tdata.ext_data.solid_velocity(2);
213 VecList fluid_vel(tdata.num_vel_pts_blade);
214 // Loop through each blade and apply the FLLC
215 for (int ib = 0; ib < tdata.num_blades; ++ib) {
216 auto& blade = tdata.blades[ib];
217 auto& fllc = tdata.fllc[ib];
218 if (!fllc.initialized) {
219 continue;
220 }
221 // Same first points as the blade views (make_component_views)
222 const int start = (ib * tdata.num_pts_blade) + 1;
223 const int start_vel = (ib * tdata.num_vel_pts_blade) + 1;
224 for (int i = 0; i < tdata.num_vel_pts_blade; ++i) {
225 fluid_vel[i][0] = uvel[start_vel + i];
226 fluid_vel[i][1] = vvel[start_vel + i];
227 fluid_vel[i][2] = wvel[start_vel + i];
228 }
229 for (int i = 0; i < tdata.num_pts_blade; ++i) {
230 const vs::Vector solid_vel{
231 static_cast<amrex::Real>(xdot[start + i]),
232 static_cast<amrex::Real>(ydot[start + i]),
233 static_cast<amrex::Real>(zdot[start + i])};
234 if (fllc.mismatched_points) {
235 blade.vel_rel[i] = interp::linear(
236 fllc.span_distance_vel, fluid_vel,
237 fllc.span_distance_force[i]) -
238 solid_vel;
239 } else {
240 blade.vel_rel[i] = fluid_vel[i] - solid_vel;
241 }
242 }
243 FLLCOp()(blade, fllc);
244 }
245 }
246
247 for (int i = 0; i < tdata.ext_data.length_fluid_velocity(0); ++i) {
248 uvel[i] = static_cast<float>(vel[i].x());
249 vvel[i] = static_cast<float>(vel[i].y());
250 wvel[i] = static_cast<float>(vel[i].z());
251 }
252}
