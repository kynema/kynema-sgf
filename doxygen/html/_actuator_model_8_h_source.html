235template <typename ActTrait, typename SrcTrait>
236void ActModel<ActTrait, SrcTrait>::determine_root_proc(
237 amrex::Vector<int>& act_proc_count)
238{
239 ops::determine_root_proc<ActTrait>(m_data, act_proc_count);
240 {
241 // Sanity checks
242 const auto& linfo = m_data.info();
243 const auto& plist = linfo.procs;
244 AMREX_ALWAYS_ASSERT_WITH_MESSAGE(
245 linfo.root_proc > -1,
246 "linfo.root_proc = " + std::to_string(linfo.root_proc));
247 AMREX_ALWAYS_ASSERT_WITH_MESSAGE(
248 plist.find(linfo.root_proc) != plist.end(),
249 "linfo.root_proc = " + std::to_string(linfo.root_proc) +
250 ", plist.size() = " + std::to_string(plist.size()));
251 }
252}
