21
22 pp.get("openfast_start_time", tf.start_time);
23 pp.get("openfast_stop_time", tf.stop_time);
24

26 pp.query("openfast_sim_mode", sim_mode);
27
28 if (sim_mode == "init") {
29 tf.sim_mode = ::ext_turb::SimMode::init;
30 amrex::Print() << "Initializing turbine:" << tf.tlabel << std::endl;
31 } else if (sim_mode == "replay") {
32 tf.sim_mode = ::ext_turb::SimMode::replay;
33 amrex::Print() << "Replaying turbine:" << tf.tlabel << std::endl;
34 } else if (sim_mode == "restart") {
35 tf.sim_mode = ::ext_turb::SimMode::restart;
36 amrex::Print() << "Restarting turbine:" << tf.tlabel << std::endl;
37 } else {
38 amrex::Abort(
39 "Actuator: Invalid OpenFAST simulation mode: " + sim_mode);
40 }
41
42 // If we are using OpenFAST restart file, require that the user provide
43 // the path to the checkpoint file.
44 if (tf.sim_mode == ::ext_turb::SimMode::restart) {
45 pp.get("openfast_restart_file", tf.checkpoint_file);
46 } else {
47 pp.get("openfast_input_file", tf.input_file);
48 }
49
50 // Check if user wants to override density checks
51 bool override_density_check = false;
52 pp.query("override_density_check", override_density_check);
53 // Check if simulation is a restart
54 bool is_restart = data.sim().io_manager().is_restart();

72 // Check other density parsing arguments against turbine air density
73 void perform_density_checks(
74 const amrex::Real rho_tdef, const bool no_check, const bool is_rst)
75 {
76 // Check if other arguments are in input file
77 amrex::ParmParse pp_incflo("incflo");
78 amrex::ParmParse pp_cstdns("ConstValue.density");
79 bool is_incflo = pp_incflo.contains("density");
80 bool is_cstdns = pp_cstdns.contains("value");
81 // Flags for detecting conflicts
82 bool cft_incflo = false;
83 bool cft_cstdns = false;
84
85 // Do warnings or aborts depending on values
86 const amrex::Real tiny = std::numeric_limits<amrex::Real>::epsilon();
87 if (is_incflo) {
88 amrex::Real rho_incflo = 1.0_rt;
89 pp_incflo.query("density", rho_incflo);
90 if (std::abs(rho_incflo - rho_tdef) > tiny) {
91 cft_incflo = true;
92 if (!no_check) {
93 amrex::Abort(
94 "Density conflict detected between TurbineFast "
95 "density and incflo.density.\n----- Values are " +
96 std::to_string(rho_tdef) + " and " +
97 std::to_string(rho_incflo) +
98 ", respectively. Check the problem setup and the "
99 "turbine definition.\n----- If this difference is "
100 "intended, set the override_density_check flag to "
101 "true.\n");
102 } else {
103 amrex::Print()
104 << "WARNING: Density conflict detected between FAST "
105 "Turbine density and incflo.density.\n----- Values "
106 "are " +
107 std::to_string(rho_tdef) + " and " +
108 std::to_string(rho_incflo) +
109 ", respectively. Abort overridden by flag.\n";
110 }
111 }
112 }
113 if (is_cstdns) {
114 amrex::Real rho_cstdns = 1.0_rt;
115 pp_cstdns.query("value", rho_cstdns);
116 if (std::abs(rho_cstdns - rho_tdef) > tiny) {
117 cft_cstdns = true;
118 if (!no_check) {
119 amrex::Abort(
120 "Density conflict detected between TurbineFast "
121 "density and ConstValue.density.value.\n----- Values "
122 "are " +
123 std::to_string(rho_tdef) + " and " +
124 std::to_string(rho_cstdns) +
125 ", respectively. Check the problem setup and the "
126 "turbine definition.\n----- If this difference is "
127 "intended, set the override_density_check flag to "
128 "true.\n");
129 } else {
130 amrex::Print()
131 << "WARNING: Density conflict detected between "
132 "TurbineFast "
133 "density and "
134 "ConstValue.density.value.\n----- Values are " +
135 std::to_string(rho_tdef) + " and " +
136 std::to_string(rho_cstdns) +
137 ", respectively. Abort overridden by flag.\n";
138 }
139 }
140 }
141 if (!is_incflo && !is_cstdns) {
142 amrex::Print()
143 << "TurbineFast: no density conflict detected, but no density "
144 "arguments found to check against.\n";
145 } else if (!cft_incflo && !cft_cstdns) {
146 amrex::Print() << "TurbineFast: no density conflict detected.\n";
147 }
148 if (is_rst) {
149 amrex::Print()
150 << "TurbineFast: simulation begins using a checkpoint file, "
151 "density compatibility must be manually confirmed between "
152 "the precursor simulation and the specified TurbineFast "
153 "density, " +
154 std::to_string(rho_tdef)
155 << ".\n";
156 }
157 }

192 void check_fast_sim_time(TurbineFast::DataType& data)
193 {
194 const auto& time = data.sim().time();
195
196 // Set OpenFAST solver end time to be at least as long as the CFD time.
197 // User can choose a longer duration in input file.
198 const amrex::Real stop1 = time.stop_time() > 0.0_rt
199 ? time.stop_time()
200 : std::numeric_limits<amrex::Real>::max();
201 const amrex::Real stop2 = time.stop_time_index() > -1
202 ? time.stop_time_index() * time.delta_t()
203 : std::numeric_limits<amrex::Real>::max();
204 const amrex::Real cfd_stop = amrex::min(stop1, stop2);
205 const amrex::Real cfd_start = time.current_time();
206 const amrex::Real cfd_sim = cfd_stop - cfd_start - 1.0e-6_rt;
207
208 // Ensure that the user specified stop_time is not shorter than CFD sim
209 const auto& tf = data.meta().ext_data;
210 const amrex::Real fast_sim = tf.stop_time - tf.start_time;
211 AMREX_ALWAYS_ASSERT_WITH_MESSAGE(
212 fast_sim > cfd_sim,
213 "OpenFAST simulation time is shorter than Kynema-SGF duration");
214 }
