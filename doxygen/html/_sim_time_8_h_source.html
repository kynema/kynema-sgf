15AMREX_FORCE_INLINE amrex::Real get_enforced_dt_for_output(
16 const amrex::Real dt,
17 const amrex::Real cur_time,
18 const amrex::Real interval,
19 const amrex::Real tol)
20{
21 // Tolerance for being slightly under is relative to requested interval
22 return amrex::min<amrex::Real>(
23 dt, ((std::floor((cur_time / interval) + tol) + 1.0_rt) * interval) -
24 cur_time);
25}
